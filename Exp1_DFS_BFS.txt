"""
EXPERIMENT 1 : DEPTH FIRST SEARCH (DFS) and BREADTH FIRST SEARCH (BFS)
========================================================================
Core idea to remember (works for ANY variation they give you):
  - DFS = go as DEEP as possible first, then backtrack  -> uses a STACK (or recursion)
  - BFS = explore LEVEL BY LEVEL                          -> uses a QUEUE

If the exam gives you a different graph, you ONLY need to change the
'graph' dictionary below. The dfs()/bfs() functions never need to change.
"""

# ============================================================
# PART A: ORIGINAL VERSION (matches practical file style)
# ============================================================

graph = {
    'A': ['B', 'C'],
    'B': ['D', 'E'],
    'C': ['F'],
    'D': [],
    'E': ['F'],
    'F': []
}

def dfs(graph, start, visited=None):
    if visited is None:
        visited = []
    visited.append(start)
    for neighbor in graph[start]:
        if neighbor not in visited:
            dfs(graph, neighbor, visited)
    return visited

from collections import deque

def bfs(graph, start):
    visited = [start]
    queue = deque([start])
    while queue:
        node = queue.popleft()
        for neighbor in graph[node]:
            if neighbor not in visited:
                visited.append(neighbor)
                queue.append(neighbor)
    return visited

print("===== PART A: ORIGINAL =====")
print("DFS Traversal:", dfs(graph, 'A'))
print("BFS Traversal:", bfs(graph, 'A'))


# ============================================================
# PART B: ALTERNATE VERSION
#   - Different graph
#   - DFS done ITERATIVELY (no recursion) using an explicit stack
#   - BFS modified to also print the PATH to a target node
# ============================================================

graph2 = {
    'S': ['A', 'B'],
    'A': ['C', 'D'],
    'B': ['E'],
    'C': [],
    'D': ['F'],
    'E': ['F'],
    'F': []
}

def dfs_iterative(graph, start):
    visited = []
    stack = [start]
    while stack:
        node = stack.pop()              # pop from the END = LIFO = DFS behaviour
        if node not in visited:
            visited.append(node)
            # reversed() keeps left-to-right neighbor order like recursive DFS
            for neighbor in reversed(graph[node]):
                if neighbor not in visited:
                    stack.append(neighbor)
    return visited

def bfs_with_path(graph, start, target):
    visited = [start]
    queue = deque([[start]])            # store whole paths, not just nodes
    while queue:
        path = queue.popleft()
        node = path[-1]
        if node == target:
            return path
        for neighbor in graph[node]:
            if neighbor not in visited:
                visited.append(neighbor)
                queue.append(path + [neighbor])
    return None

print("\n===== PART B: ALTERNATE (different graph + variations) =====")
print("Iterative DFS Traversal:", dfs_iterative(graph2, 'S'))
print("BFS Shortest Path S -> F:", bfs_with_path(graph2, 'S', 'F'))


# ============================================================
# PART C: EASY / BEGINNER VERSION with line-by-line explanation
# ============================================================
"""
Think of the graph as a dictionary:  key = node, value = list of its neighbours.

DFS LOGIC (step by step):
  1. Start at a node, mark it visited, print/record it.
  2. Look at its neighbours one by one.
  3. If a neighbour hasn't been visited yet, go into it FIRST (recursion does this)
     before coming back to check the next neighbour.
  4. This naturally dives deep down one branch before trying siblings.

BFS LOGIC (step by step):
  1. Start at a node, mark it visited, put it in a QUEUE.
  2. Take the front item out of the queue, look at its neighbours.
  3. Any unvisited neighbour gets marked visited and added to the BACK of the queue.
  4. Repeat until the queue is empty.
  5. Because we always add to the back and remove from the front (FIFO),
     we finish ALL nodes at one level before moving deeper.
"""

simple_graph = {
    1: [2, 3],
    2: [4],
    3: [4],
    4: []
}

def simple_dfs(g, node, visited):
    if node not in visited:
        print("Visiting:", node)      # do the "visit" action
        visited.append(node)
        for next_node in g[node]:     # go through each neighbour
            simple_dfs(g, next_node, visited)   # recursive call = going deeper

def simple_bfs(g, start):
    visited = [start]
    queue = [start]                   # using a plain list as a queue
    while queue:
        current = queue.pop(0)        # remove from FRONT (index 0) = FIFO
        print("Visiting:", current)
        for neighbour in g[current]:
            if neighbour not in visited:
                visited.append(neighbour)
                queue.append(neighbour)   # add to BACK

print("\n===== PART C: EASY VERSION (with prints to show the logic) =====")
print("-- DFS --")
simple_dfs(simple_graph, 1, [])
print("-- BFS --")
simple_bfs(simple_graph, 1)
